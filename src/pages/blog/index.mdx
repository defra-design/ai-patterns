import GridRow from '../../components/grid/GridRow.astro'
import GridColumn from '../../components/grid/GridColumn.astro'
import { getBlogPosts } from '../../utils/patterns.js'

export const sortedPosts = getBlogPosts(import.meta.glob('./*.md', { eager: true }))

<GridRow>
    <GridColumn width="two-thirds">
        # Blog

        <p class="govuk-body-l">
            Sharing our experiences implementing AI solutions, lessons learned, and guidance for teams working with artificial intelligence in government services.
        </p>

        <div class="blog-posts">
            {sortedPosts.length > 0 ? (
                sortedPosts.map(post => (
                    <article class="blog-post">
                        <h2 class="govuk-heading-m govuk-!-margin-bottom-2">
                            <a href={post.url} class="govuk-link">{post.frontmatter.title}</a>
                        </h2>
                        {post.frontmatter.date && (
                            <p class="govuk-body-s blog-meta">
                                Posted on <time datetime={post.frontmatter.date}>
                                    {new Date(post.frontmatter.date).toLocaleDateString('en-GB', { 
                                        day: 'numeric', 
                                        month: 'long', 
                                        year: 'numeric' 
                                    })}
                                </time>
                            </p>
                        )}
                        {post.frontmatter.description && (
                            <p class="govuk-body">{post.frontmatter.description}</p>
                        )}
                        <a href={post.url} class="govuk-link">Read this post</a>
                    </article>
                ))
            ) : (
                <p class="govuk-body govuk-!-margin-top-6">
                    We are currently developing blog posts for this section. Check back soon for updates.
                </p>
            )}
        </div>
    </GridColumn>
</GridRow>
